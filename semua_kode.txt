// src-tauri/src/avatar.rs
// Modul Pengelola Custom Avatar Sprite 2D & Sistem Multi-Preset
// Menyimpan, membaca, mengekstrak ZIP, dan mengelola preset karakter di:
//   ~/.config/aoi/avatar_presets/<preset_id>/
// Mendukung format PNG, WebP, SVG, JPG.

use std::collections::HashMap;
use std::fs;
use std::path::{Path, PathBuf};
use std::process::Command;
use base64::prelude::*;
use serde::{Deserialize, Serialize};

pub const FRAME_NAMES: &[&str] = &[
    "idle_open",
    "idle_blink",
    "listening",
    "thinking",
    "speak_open",
    "speak_closed",
    "confirm",
    "error",
];

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AvatarPresetMeta {
    pub id: String,
    pub name: String,
    pub thumbnail: String, // Data URL base64 dari frame idle_open atau frame pertama
    pub frame_count: usize,
    pub is_active: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct PresetManifest {
    pub id: String,
    pub name: String,
    #[serde(default)]
    pub author: String,
    #[serde(default)]
    pub frames: HashMap<String, String>, // nama_frame -> nama_file atau data_url
}

pub fn get_legacy_avatar_dir() -> PathBuf {
    let home = std::env::var("HOME").unwrap_or_else(|_| "/tmp".into());
    PathBuf::from(home).join(".config/aoi/avatar")
}

pub fn get_presets_root_dir() -> PathBuf {
    let home = std::env::var("HOME").unwrap_or_else(|_| "/tmp".into());
    PathBuf::from(home).join(".config/aoi/avatar_presets")
}

pub fn get_preset_dir(preset_id: &str) -> PathBuf {
    get_presets_root_dir().join(preset_id)
}

/// Inisialisasi struktur folder presets dan migrasi data legacy avatar jika ada
pub fn ensure_presets_initialized() {
    let root = get_presets_root_dir();
    let _ = fs::create_dir_all(&root);

    // Cek apakah ada avatar lama di ~/.config/aoi/avatar/ yang belum dimigrasi
    let legacy_dir = get_legacy_avatar_dir();
    let migrated_preset_dir = root.join("custom_migrated");

    if legacy_dir.exists() && !migrated_preset_dir.exists() {
        // Cek apakah legacy_dir memiliki gambar
        let mut has_images = false;
        if let Ok(entries) = fs::read_dir(&legacy_dir) {
            for entry in entries.flatten() {
                if let Some(ext) = entry.path().extension().and_then(|e| e.to_str()) {
                    if ["png", "webp", "jpg", "jpeg", "svg"].contains(&ext.to_lowercase().as_str()) {
                        has_images = true;
                        break;
                    }
                }
            }
        }

        if has_images {
            let _ = fs::create_dir_all(&migrated_preset_dir);
            // Salin semua file dari legacy_dir ke migrated_preset_dir
            if let Ok(entries) = fs::read_dir(&legacy_dir) {
                for entry in entries.flatten() {
                    let dest = migrated_preset_dir.join(entry.file_name());
                    let _ = fs::copy(entry.path(), dest);
                }
            }

            let manifest = PresetManifest {
                id: "custom_migrated".to_string(),
                name: "Avatar Kustom Saya".to_string(),
                author: "User".to_string(),
                frames: HashMap::new(),
            };
            let _ = save_preset_manifest(&migrated_preset_dir, &manifest);

            // Set preset aktif ke custom_migrated jika config masih default
            let mut cfg = crate::config::load_config();
            if cfg.active_avatar_preset == "default" {
                cfg.active_avatar_preset = "custom_migrated".to_string();
                let _ = crate::config::save_config(&cfg);
            }
        }
    }
}

/// Dapatkan daftar semua preset avatar yang tersedia
pub fn list_avatar_presets() -> Result<Vec<AvatarPresetMeta>, String> {
    ensure_presets_initialized();
    let root = get_presets_root_dir();
    let cfg = crate::config::load_config();
    let active_id = cfg.active_avatar_preset;

    let mut list = Vec::new();

    // 1. Preset Bawaan: Anime SVG
    list.push(AvatarPresetMeta {
        id: "default".to_string(),
        name: "Aoi Classic (Anime SVG)".to_string(),
        thumbnail: "".to_string(), // Frontend menampilkan thumbnail SVG built-in
        frame_count: 8,
        is_active: active_id == "default",
    });

    // 2. Scan folder presets di ~/.config/aoi/avatar_presets/*
    if let Ok(entries) = fs::read_dir(&root) {
        let mut subdirs: Vec<PathBuf> = entries
            .flatten()
            .map(|e| e.path())
            .filter(|p| p.is_dir() && p.file_name().and_then(|s| s.to_str()) != Some("default"))
            .collect();
        subdirs.sort();

        for dir in subdirs {
            let folder_name = dir.file_name().unwrap_or_default().to_string_lossy().to_string();
            let manifest = load_preset_manifest(&dir);
            let display_name = if !manifest.name.trim().is_empty() {
                manifest.name
            } else {
                folder_name.clone()
            };

            // Hitung frame yang tersedia dan ambil thumbnail
            let pack = load_preset_frames_from_dir(&dir);
            let count = pack.len();
            let thumb = pack.get("idle_open")
                .or_else(|| pack.get("idle_blink"))
                .or_else(|| pack.values().next())
                .cloned()
                .unwrap_or_default();

            list.push(AvatarPresetMeta {
                id: folder_name.clone(),
                name: display_name,
                thumbnail: thumb,
                frame_count: count,
                is_active: active_id == folder_name,
            });
        }
    }

    Ok(list)
}

/// Aktifkan salah satu preset avatar dan muat frame-nya
pub fn switch_avatar_preset(preset_id: &str) -> Result<HashMap<String, String>, String> {
    let mut cfg = crate::config::load_config();
    cfg.active_avatar_preset = preset_id.to_string();
    let _ = crate::config::save_config(&cfg);

    if preset_id == "default" {
        return Ok(HashMap::new());
    }

    let dir = get_preset_dir(preset_id);
    if !dir.exists() {
        return Err(format!("Preset '{}' tidak ditemukan.", preset_id));
    }

    Ok(load_preset_frames_from_dir(&dir))
}

/// Buat preset baru yang kosong dengan nama tertentu
pub fn create_avatar_preset(name: &str) -> Result<AvatarPresetMeta, String> {
    ensure_presets_initialized();
    let timestamp = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_secs())
        .unwrap_or(0);

    let clean_id = format!("preset_{}", timestamp);
    let dir = get_preset_dir(&clean_id);
    fs::create_dir_all(&dir)
        .map_err(|e| format!("Gagal membuat folder preset: {}", e))?;

    let manifest = PresetManifest {
        id: clean_id.clone(),
        name: if name.trim().is_empty() { "Preset Baru".to_string() } else { name.trim().to_string() },
        author: "User".to_string(),
        frames: HashMap::new(),
    };
    save_preset_manifest(&dir, &manifest)?;

    Ok(AvatarPresetMeta {
        id: clean_id,
        name: manifest.name,
        thumbnail: "".to_string(),
        frame_count: 0,
        is_active: false,
    })
}

/// Hapus preset avatar
pub fn delete_avatar_preset(preset_id: &str) -> Result<(), String> {
    if preset_id == "default" {
        return Err("Preset bawaan tidak dapat dihapus.".to_string());
    }

    let dir = get_preset_dir(preset_id);
    if dir.exists() {
        fs::remove_dir_all(&dir)
            .map_err(|e| format!("Gagal menghapus folder preset: {}", e))?;
    }

    // Jika yang dihapus sedang aktif, kembalikan ke default
    let mut cfg = crate::config::load_config();
    if cfg.active_avatar_preset == preset_id {
        cfg.active_avatar_preset = "default".to_string();
        let _ = crate::config::save_config(&cfg);
    }

    Ok(())
}

/// Impor berkas ZIP yang di-upload dari UI (base64 encoded) menjadi preset avatar baru
pub fn import_avatar_preset_zip(name: Option<String>, zip_base64: &str) -> Result<AvatarPresetMeta, String> {
    ensure_presets_initialized();

    // Decode zip bytes
    let clean_b64 = if let Some(idx) = zip_base64.find(";base64,") {
        &zip_base64[idx + 8..]
    } else {
        zip_base64
    };

    let bytes = BASE64_STANDARD
        .decode(clean_b64.trim())
        .map_err(|e| format!("Gagal decode base64 file zip: {}", e))?;

    let timestamp = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_secs())
        .unwrap_or(0);

    let temp_zip_path = format!("/tmp/aoi_upload_{}.zip", timestamp);
    fs::write(&temp_zip_path, bytes)
        .map_err(|e| format!("Gagal menulis berkas ZIP sementara: {}", e))?;

    let preset_id = format!("preset_{}", timestamp);
    let target_dir = get_preset_dir(&preset_id);
    let _ = fs::create_dir_all(&target_dir);

    // Ekstrak via unzip CLI
    let status = Command::new("unzip")
        .args(["-o", "-q", &temp_zip_path, "-d", target_dir.to_str().unwrap_or("")])
        .status()
        .map_err(|e| format!("Gagal menjalankan unzip: {}", e))?;

    let _ = fs::remove_file(&temp_zip_path);

    if !status.success() {
        let _ = fs::remove_dir_all(&target_dir);
        return Err("Gagal mengekstrak berkas ZIP. Pastikan format arsip valid.".to_string());
    }

    // Tangani kemungkinan file bersarang di dalam subfolder (misal saat zip folder)
    flatten_dir_if_nested(&target_dir);

    // Normalisasi nama frame gambar (misal jika ada file 'idle.png' -> 'idle_open.png')
    normalize_frame_files(&target_dir);

    let preset_name = name
        .filter(|n| !n.trim().is_empty())
        .unwrap_or_else(|| format!("Preset {}", timestamp));

    let manifest = PresetManifest {
        id: preset_id.clone(),
        name: preset_name.clone(),
        author: "User Import".to_string(),
        frames: HashMap::new(),
    };
    let _ = save_preset_manifest(&target_dir, &manifest);

    // Muat frames untuk metadata
    let pack = load_preset_frames_from_dir(&target_dir);
    let count = pack.len();
    let thumb = pack.get("idle_open")
        .or_else(|| pack.get("idle_blink"))
        .or_else(|| pack.values().next())
        .cloned()
        .unwrap_or_default();

    // Otomatis aktifkan preset baru ini
    let mut cfg = crate::config::load_config();
    cfg.active_avatar_preset = preset_id.clone();
    let _ = crate::config::save_config(&cfg);

    Ok(AvatarPresetMeta {
        id: preset_id,
        name: preset_name,
        thumbnail: thumb,
        frame_count: count,
        is_active: true,
    })
}

/// Ekspor preset avatar ke format ZIP dan kembalikan base64 string
pub fn export_avatar_preset_zip(preset_id: &str) -> Result<String, String> {
    let dir = get_preset_dir(preset_id);
    if !dir.exists() {
        return Err(format!("Preset '{}' tidak ditemukan.", preset_id));
    }

    let timestamp = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_secs())
        .unwrap_or(0);

    let temp_zip_path = format!("/tmp/aoi_export_{}_{}.zip", preset_id, timestamp);

    // Eksekusi zip command (-j = flatten paths)
    let status = Command::new("zip")
        .args(["-j", "-q", &temp_zip_path])
        .args(
            FRAME_NAMES
                .iter()
                .map(|f| format!("{}/{}*", dir.to_str().unwrap_or(""), f)),
        )
        .status();

    if status.is_err() || !status.as_ref().unwrap().success() {
        // Coba zip seluruh folder jika pattern di atas kosong
        let _ = Command::new("zip")
            .args(["-r", "-j", "-q", &temp_zip_path, dir.to_str().unwrap_or("")])
            .status()
            .map_err(|e| format!("Gagal membuat berkas zip: {}", e))?;
    }

    let zip_bytes = fs::read(&temp_zip_path)
        .map_err(|e| format!("Gagal membaca hasil zip: {}", e))?;
    let _ = fs::remove_file(&temp_zip_path);

    Ok(BASE64_STANDARD.encode(&zip_bytes))
}

pub fn save_avatar_sprite(frame_name: &str, data_url: &str) -> Result<String, String> {
    if !FRAME_NAMES.contains(&frame_name) {
        return Err(format!("Nama frame '{}' tidak valid.", frame_name));
    }

    let dir = get_avatar_dir();
    fs::create_dir_all(&dir)
        .map_err(|e| format!("Gagal membuat direktori avatar {:?}: {}", dir, e))?;

    let (mime, raw_b64) = if let Some(idx) = data_url.find(";base64,") {
        let prefix = &data_url[..idx];
        let mime = prefix.strip_prefix("data:").unwrap_or("image/png");
        let b64 = &data_url[idx + 8..];
        (mime, b64)
    } else {
        ("image/png", data_url)
    };

    let ext = match mime {
        "image/webp" => "webp",
        "image/jpeg" | "image/jpg" => "jpg",
        "image/svg+xml" => "svg",
        _ => "png",
    };

    let bytes = BASE64_STANDARD
        .decode(raw_b64.trim())
        .map_err(|e| format!("Gagal decode base64: {}", e))?;

    let file_path = dir.join(format!("{}.{}", frame_name, ext));
    fs::write(&file_path, bytes)
        .map_err(|e| format!("Gagal menulis file avatar {:?}: {}", file_path, e))?;

    let mut manifest = load_manifest_map(&dir);
    manifest.insert(frame_name.to_string(), data_url.to_string());
    save_manifest_map(&dir, &manifest)?;

    Ok(data_url.to_string())
}

pub fn save_avatar_pack(pack: HashMap<String, String>) -> Result<(), String> {
    let dir = get_avatar_dir();
    fs::create_dir_all(&dir)
        .map_err(|e| format!("Gagal membuat direktori avatar: {}", e))?;

    for (k, v) in &pack {
        if !v.is_empty() {
            let _ = save_avatar_sprite(k, v);
        }
    }
    Ok(())
}

pub fn load_avatar_pack(folder_override: Option<String>) -> Result<HashMap<String, String>, String> {
    let dir = if let Some(ref path_str) = folder_override {
        if !path_str.trim().is_empty() {
            PathBuf::from(path_str.trim())
        } else {
            get_avatar_dir()
        }
    } else {
        let cfg = crate::config::load_config();
        if cfg.active_avatar_preset == "default" {
            return Ok(HashMap::new());
        }
        get_preset_dir(&cfg.active_avatar_preset)
    };

    if !dir.exists() {
        return Ok(HashMap::new());
    }

    Ok(load_preset_frames_from_dir(&dir))
}

pub fn reset_avatar_pack() -> Result<(), String> {
    let mut cfg = crate::config::load_config();
    cfg.active_avatar_preset = "default".to_string();
    let _ = crate::config::save_config(&cfg);
    Ok(())
}

fn load_preset_frames_from_dir(dir: &Path) -> HashMap<String, String> {
    // 1. Cek manifest.json
    let manifest_map = load_manifest_map(dir);
    if !manifest_map.is_empty() {
        return manifest_map;
    }

    // 2. Scan disk files
    let mut map = HashMap::new();
    let extensions = ["png", "webp", "jpg", "jpeg", "svg"];

    for &frame in FRAME_NAMES {
        for &ext in &extensions {
            let candidate = dir.join(format!("{}.{}", frame, ext));
            if candidate.exists() {
                if let Ok(bytes) = fs::read(&candidate) {
                    let mime = match ext {
                        "webp" => "image/webp",
                        "jpg" | "jpeg" => "image/jpeg",
                        "svg" => "image/svg+xml",
                        _ => "image/png",
                    };
                    let b64 = BASE64_STANDARD.encode(&bytes);
                    let data_url = format!("data:{};base64,{}", mime, b64);
                    map.insert(frame.to_string(), data_url);
                    break;
                }
            }
        }
    }

    map
}

fn load_manifest_map(dir: &Path) -> HashMap<String, String> {
    let path = dir.join("manifest.json");
    if path.exists() {
        if let Ok(content) = fs::read_to_string(&path) {
            // Bisa format Map<String, String> atau Struct PresetManifest
            if let Ok(map) = serde_json::from_str::<HashMap<String, String>>(&content) {
                return map;
            }
            if let Ok(manifest) = serde_json::from_str::<PresetManifest>(&content) {
                return manifest.frames;
            }
        }
    }
    HashMap::new()
}

fn save_manifest_map(dir: &Path, map: &HashMap<String, String>) -> Result<(), String> {
    let path = dir.join("manifest.json");
    let json_str = serde_json::to_string_pretty(map)
        .map_err(|e| format!("Gagal serialisasi manifest: {}", e))?;
    fs::write(&path, json_str)
        .map_err(|e| format!("Gagal menulis manifest: {}", e))?;
    Ok(())
}

fn load_preset_manifest(dir: &Path) -> PresetManifest {
    let path = dir.join("manifest.json");
    if path.exists() {
        if let Ok(content) = fs::read_to_string(&path) {
            if let Ok(m) = serde_json::from_str::<PresetManifest>(&content) {
                return m;
            }
        }
    }
    PresetManifest::default()
}

fn save_preset_manifest(dir: &Path, manifest: &PresetManifest) -> Result<(), String> {
    let path = dir.join("manifest.json");
    let json_str = serde_json::to_string_pretty(manifest)
        .map_err(|e| format!("Gagal serialisasi manifest: {}", e))?;
    fs::write(&path, json_str)
        .map_err(|e| format!("Gagal menulis manifest: {}", e))?;
    Ok(())
}

fn flatten_dir_if_nested(target_dir: &Path) {
    if let Ok(entries) = fs::read_dir(target_dir) {
        let items: Vec<PathBuf> = entries.flatten().map(|e| e.path()).collect();
        // Jika hanya ada 1 direktori di dalam target_dir (e.g. user zip folder "my_waifu/")
        if items.len() == 1 && items[0].is_dir() {
            let nested = &items[0];
            if let Ok(sub_entries) = fs::read_dir(nested) {
                for sub in sub_entries.flatten() {
                    let dest = target_dir.join(sub.file_name());
                    let _ = fs::rename(sub.path(), dest);
                }
            }
            let _ = fs::remove_dir(nested);
        }
    }
}

fn normalize_frame_files(target_dir: &Path) {
    let aliases: &[(&[&str], &str)] = &[
        (&["idle", "stand", "normal"], "idle_open"),
        (&["blink", "kedip", "close_eyes"], "idle_blink"),
        (&["listen", "hear", "dengar", "mic"], "listening"),
        (&["think", "pikir", "wait"], "thinking"),
        (&["speak", "talking", "bicara", "say"], "speak_open"),
        (&["mouth_closed", "speak_close", "bicara_tutup"], "speak_closed"),
        (&["ok", "yes", "success", "alert", "warn"], "confirm"),
        (&["err", "fail", "angry", "marah"], "error"),
    ];

    if let Ok(entries) = fs::read_dir(target_dir) {
        for entry in entries.flatten() {
            let p = entry.path();
            if !p.is_file() {
                continue;
            }
            let stem = p.file_stem().and_then(|s| s.to_str()).unwrap_or("").to_lowercase();
            let ext = p.extension().and_then(|s| s.to_str()).unwrap_or("png");

            // Jika belum punya nama frame standar, cek apakah cocok dengan alias
            if !FRAME_NAMES.contains(&stem.as_str()) {
                for (keywords, target_name) in aliases {
                    if keywords.iter().any(|&k| stem == k || stem.starts_with(k)) {
                        let new_dest = target_dir.join(format!("{}.{}", target_name, ext));
                        if !new_dest.exists() {
                            let _ = fs::rename(&p, &new_dest);
                        }
                        break;
                    }
                }
            }
        }
    }
}

// src-tauri/src/bash_executor.rs
// Two-Tier Command Execution System
//
// ┌─────────────────────────────────────────────────────────────────────┐
// │  GREEN TIER  → Aman, langsung dieksekusi tanpa konfirmasi           │
// │  RED TIER    → Destruktif/berbahaya, wajib konfirmasi Alt+Y/Alt+N  │
// └─────────────────────────────────────────────────────────────────────┘
//
// Regex classifier memfilter kata kunci berbahaya SEBELUM menyentuh shell.
// Semua eksekusi menggunakan `sh -c` dengan timeout 30 detik.

use std::process::Stdio;
use tokio::process::Command;
use tokio::time::{timeout, Duration};

// ────────────────────────────────────────────────────────────────────────────
// Classifier: Tentukan tier dari sebuah perintah
// ────────────────────────────────────────────────────────────────────────────

/// Kembalikan `true` jika perintah termasuk Red Tier (butuh konfirmasi)
pub fn is_red_tier(command: &str) -> bool {
    // Kata kunci yang wajib memicu konfirmasi pengguna
    let red_patterns = [
        r"sudo\b",
        r"\brm\b",
        r"\bmv\b",
        r"\bdd\b",
        r"\bmkfs\b",
        r"\bformat\b",
        r"\bfdisk\b",
        r"\bparted\b",
        r"\bshred\b",
        r"\bpacman\b",
        r"\bparu\b",
        r"\byay\b",
        r"\bsystemctl\b",
        r"\bjournalctl\b",
        r"\bkill\b",
        r"\bpkill\b",
        r"\bkillall\b",
        r"\bchmod\b",
        r"\bchown\b",
        r"\bchattr\b",
        r"\biptables\b",
        r"\bufw\b",
        r"\bcurl\b.*\|\s*bash",   // curl | bash — sangat berbahaya
        r"\bwget\b.*\|\s*sh",     // wget | sh
        r">\s*/dev/",             // redirect ke /dev/
        r">\s*/etc/",             // overwrite config sistem
        r">\s*/boot/",
        r">\s*/usr/",
        r"\btruncate\b",
        r"--no-preserve-root",
    ];

    for pattern in &red_patterns {
        if let Ok(re) = regex_lite(pattern, command) {
            if re {
                return true;
            }
        }
    }
    false
}

/// Cek apakah `command` mengandung pattern (manual regex-lite tanpa dep regex)
fn regex_lite(pattern: &str, text: &str) -> Result<bool, ()> {
    // Implementasi sederhana: strip \b dan cek kata
    // Untuk production, tambahkan crate `regex` ke Cargo.toml
    let core = pattern
        .trim_start_matches(r"\b")
        .trim_end_matches(r"\b")
        .replace(r"\b", " ")
        .replace(r"\s*", " ")
        .replace(r".*", "");

    // Handle pattern yang mengandung | (pipe patterns)
    if pattern.contains(r"\|") {
        // curl | bash, wget | sh — cek substring kombinasi
        if text.contains("curl") && text.contains("bash") { return Ok(true); }
        if text.contains("wget") && text.contains("sh") { return Ok(true); }
        return Ok(false);
    }

    // Handle redirect patterns
    if pattern.starts_with('>') {
        let target = core.trim_start_matches("> /").trim_start_matches(">/");
        return Ok(text.contains(&format!("/{}", target)) || text.contains(&format!("> /{}", target)));
    }

    // Cek apakah kata kunci muncul sebagai kata penuh (dikelilingi spasi/awal/akhir)
    let word = core.trim();
    let found = text.split_whitespace().any(|w| {
        let w = w.trim_matches(|c: char| !c.is_alphanumeric() && c != '-' && c != '_');
        w == word
    }) || text.contains(&format!(" {} ", word))
        || text.starts_with(&format!("{} ", word))
        || text == word;

    Ok(found)
}

// ────────────────────────────────────────────────────────────────────────────
// Parser: Ekstrak BASH:<command> dari output LLM
// ────────────────────────────────────────────────────────────────────────────

/// Cek apakah LLM response mengandung tool call BASH:<command>
/// Format: "...teks...\nBASH:ls -la /home\n...teks..."
pub fn extract_bash_command(llm_response: &str) -> Option<String> {
    for line in llm_response.lines() {
        let line = line.trim();
        if line.starts_with("BASH:") {
            let cmd = line.trim_start_matches("BASH:").trim().to_string();
            if !cmd.is_empty() {
                return Some(cmd);
            }
        }
    }
    None
}

/// Pisahkan teks biasa dari bagian BASH: di output LLM
pub fn strip_bash_command(llm_response: &str) -> String {
    llm_response
        .lines()
        .filter(|l| !l.trim().starts_with("BASH:"))
        .collect::<Vec<_>>()
        .join("\n")
        .trim()
        .to_string()
}

/// Eksekusi perintah Green Tier langsung (tanpa konfirmasi).
/// Timeout: 30 detik. Output di-capture dan dikembalikan.
pub async fn execute_green(command: &str) -> Result<String, String> {
    let result = timeout(
        Duration::from_secs(30),
        Command::new("sh")
            .arg("-c")
            .arg(command)
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .output(),
    )
    .await;

    match result {
        Ok(Ok(output)) => {
            let stdout = String::from_utf8_lossy(&output.stdout).to_string();
            let stderr = String::from_utf8_lossy(&output.stderr).to_string();

            if output.status.success() {
                Ok(if stdout.is_empty() { "✅ Selesai (tidak ada output).".into() } else { stdout })
            } else {
                Err(if stderr.is_empty() {
                    format!("Perintah gagal dengan exit code: {:?}", output.status.code())
                } else {
                    stderr
                })
            }
        }
        Ok(Err(e)) => Err(format!("Gagal menjalankan proses: {}", e)),
        Err(_) => Err("Timeout: perintah berjalan lebih dari 30 detik.".into()),
    }
}

/// Hasil klasifikasi perintah LLM — dipakai oleh `commands.rs`
#[derive(Debug, serde::Serialize)]
pub struct CommandClassification {
    pub command: String,
    pub tier: CommandTier,
}

#[derive(Debug, serde::Serialize)]
pub enum CommandTier {
    Green,
    Red,
}

/// Klasifikasi perintah dan kembalikan tier-nya
pub fn classify(command: &str) -> CommandClassification {
    CommandClassification {
        command: command.to_string(),
        tier: if is_red_tier(command) {
            CommandTier::Red
        } else {
            CommandTier::Green
        },
    }
}

// LLM Pipeline — shared antara chat input dan input suara STT

pub async fn run_ai_pipeline(
    user_message: String,
    app: AppHandle,
    state_arc: Arc<Mutex<StatePayload>>,
    tts_arc: Arc<Mutex<Option<tokio::process::Child>>>,
) -> Result<String, String> {
    // ── 1. Transisi ke Thinking
    {
        let mut s = state_arc.lock().await;
        s.state = AssistantState::Thinking;
        s.message = "Sedang memikirkan jawaban...".into();
    }
    broadcast_state(&app, &state_arc).await;

    // ── 2. Kumpulkan konteks Hyprland (window aktif & clipboard)
    let ctx = hyprland::build_context(false).await;
    let context_str = hyprland::format_context_for_prompt(&ctx);

    // ── 3. Bangun pesan dengan konteks desktop
    let enriched_message = if context_str.trim().is_empty() {
        user_message.clone()
    } else {
        format!("{}\n\n{}", user_message, context_str)
    };

    // ── 4. Konfigurasi LLM dari file config / env
    let config = build_llm_config();

    // Gunakan tauri::Manager untuk mengambil MemoryState dari AppHandle
    let history = {
        use tauri::Manager;
        let mem_state = app.state::<crate::MemoryState>();
        let mut mem = mem_state.0.lock().await;
        // Simpan pesan asli user ke memori
        mem.push_user(user_message.clone());

        // Salin seluruh history, lalu perkaya pesan terakhir dengan konteks Hyprland
        let mut msgs = mem.messages().to_vec();
        if let Some(last) = msgs.last_mut() {
            last.content = enriched_message;
        }
        msgs
    };

    // ── 5. Panggil LLM provider
    let llm_result = llm_client::chat(&config, &history)
        .await
        .map(|resp| strip_thinking_tags(&resp));

    // Jika berhasil, simpan teks balasan bersih (tanpa tag BASH:) ke MemoryState
    if let Ok(ref response_text) = llm_result {
        use tauri::Manager;
        let mem_state = app.state::<crate::MemoryState>();
        let mut mem = mem_state.0.lock().await;
        let clean_reply = bash_executor::strip_bash_command(response_text);
        if !clean_reply.is_empty() {
            mem.push_assistant(clean_reply);
        }
    }

    match llm_result {
        Err(e) => {
            let mut s = state_arc.lock().await;
            s.state = AssistantState::Error;
            s.message = format!("❌ Error LLM: {}", e);
            drop(s);
            broadcast_state(&app, &state_arc).await;
            Err(e)
        }
        Ok(response) => {
            // ── 6. Cek tool call BASH:<perintah>
            let bash_cmd = bash_executor::extract_bash_command(&response);
            let text_reply = bash_executor::strip_bash_command(&response);

            if let Some(cmd) = bash_cmd {
                let classification = bash_executor::classify(&cmd);

                match classification.tier {
                    bash_executor::CommandTier::Green => {
                        // Langsung eksekusi (Green Tier)
                        let exec_result = bash_executor::execute_green(&cmd).await;
                        let mut s = state_arc.lock().await;
                        match exec_result {
                            Ok(output) => {
                                s.state = AssistantState::Speaking;
                                let reply = if text_reply.is_empty() {
                                    format!("✅ Selesai!\n```\n{}\n```", output.trim())
                                } else {
                                    format!("{}\n\n✅ Output:\n```\n{}\n```", text_reply, output.trim())
                                };
                                s.message = reply.clone();
                                drop(s);
                                broadcast_state(&app, &state_arc).await;

                                // Trigger TTS
                                let speech_text = if !text_reply.is_empty() {
                                    text_reply.clone()
                                } else {
                                    "Perintah berhasil dieksekusi.".to_string()
                                };
                                let app_h = app.clone();
                                let s_clone = state_arc.clone();
                                let t_clone = tts_arc.clone();
                                tokio::spawn(async move {
                                    tts::speak(speech_text, app_h, s_clone, t_clone).await;
                                });

                                Ok(reply)
                            }
                            Err(e) => {
                                s.state = AssistantState::Error;
                                s.message = format!("❌ Gagal: {}", e);
                                drop(s);
                                broadcast_state(&app, &state_arc).await;
                                Err(e)
                            }
                        }
                    }
                    bash_executor::CommandTier::Red => {
                        // Butuh konfirmasi (Red Tier)
                        let mut s = state_arc.lock().await;
                        s.state = AssistantState::Confirm;
                        s.pending_command = Some(cmd.clone());
                        s.message = if text_reply.is_empty() {
                            format!("Izinkan saya menjalankan perintah ini?")
                        } else {
                            text_reply.clone()
                        };
                        drop(s);
                        broadcast_state(&app, &state_arc).await;
                        Ok(format!(
                            "{}\n\n⚠️ Perintah memerlukan konfirmasi: `{}`",
                            text_reply, cmd
                        ))
                    }
                }
            } else {
                // ── Balasan teks biasa → Speaking + TTS + RMS Lip-Sync
                let mut s = state_arc.lock().await;
                s.state = AssistantState::Speaking;
                s.message = response.clone();
                drop(s);
                broadcast_state(&app, &state_arc).await;

                // Trigger TTS
                let speech_text = tts::clean_text_for_tts(&response);
                let app_h = app.clone();
                let s_clone = state_arc.clone();
                let t_clone = tts_arc.clone();
                tokio::spawn(async move {
                    tts::speak(speech_text, app_h, s_clone, t_clone).await;
                });

                Ok(response)
            }
        }
    }
}

#[tauri::command]
pub async fn confirm_pending_command(
    app: AppHandle,
    state: State<'_, AppState>,
    tts_state: State<'_, TtsPlayerState>,
) -> Result<(), String> {
    let pending = {
        let mut s = state.0.lock().await;
        if s.state == AssistantState::Confirm {
            let cmd = s.pending_command.take();
            s.state = AssistantState::Thinking;
            s.message = "⚡ Menjalankan perintah...".into();
            cmd
        } else {
            None
        }
    };
    broadcast_state(&app, &state.0).await;

    if let Some(bash_cmd) = pending {
        let result = bash_executor::execute_green(&bash_cmd).await;
        let mut s = state.0.lock().await;
        match result {
            Ok(out) => {
                let char_count = out.chars().count();
                let trimmed = if char_count > 1500 {
                    format!("{}...\n[Output dipotong]", out.chars().take(1500).collect::<String>())
                } else {
                    out.trim().to_string()
                };
                s.state = AssistantState::Speaking;
                s.message = format!("✅ Selesai:\n```\n{}\n```", trimmed);
                let app_h = app.clone();
                let state_arc = state.0.clone();
                let tts_arc = tts_state.0.clone();
                tokio::spawn(async move {
                    tts::speak("Perintah selesai dijalankan.".into(), app_h, state_arc, tts_arc).await;
                });
            }
            Err(e) => {
                s.state = AssistantState::Error;
                s.message = format!("❌ Error: {}", e);
            }
        }
        drop(s);
        broadcast_state(&app, &state.0).await;
    }
    Ok(())
}

// ────────────────────────────────────────────────────────────────────────────
// Helper: Membangun LlmConfig dari config.json + env fallback
// ────────────────────────────────────────────────────────────────────────────

pub fn build_llm_config() -> LlmConfig {
    let cfg = config::load_config();
    let p_lower = cfg.llm_provider.to_lowercase();

    let provider = match p_lower.as_str() {
        "openai" | "groq" | "openrouter" | "custom" => llm_client::LlmProvider::OpenAI,
        "ollama" => llm_client::LlmProvider::Ollama,
        _ => llm_client::LlmProvider::Gemini,
    };

    let api_key = match provider {
        llm_client::LlmProvider::Gemini => {
            if !cfg.gemini_api_key.is_empty() {
                Some(cfg.gemini_api_key.clone())
            } else {
                std::env::var("GEMINI_API_KEY").ok()
            }
        }
        llm_client::LlmProvider::OpenAI => {
            if p_lower == "custom" {

pub fn strip_thinking_tags(raw: &str) -> String {
    let mut text = raw.to_string();
    while let Some(start) = text.find("<thinking>") {
        if let Some(end) = text[start..].find("</thinking>") {
            let actual_end = start + end + "</thinking>".len();
            text.replace_range(start..actual_end, "");
        } else {
            text.truncate(start);
            break;
        }
    }
    if let Some(end) = text.find("</thinking>") {
        text = text[end + "</thinking>".len()..].to_string();
    }
    text.trim().to_string()
}

// Modul konteks Hyprland & Vision — membaca lingkungan desktop aktif pengguna
// untuk dikirim sebagai konteks tambahan ke LLM.
//
// Fungsi:
//   - get_active_window()  → hyprctl activewindow -j
//   - get_clipboard()      → wl-paste
//   - take_screenshot()    → grim -g <geometry> /tmp/aoi_screenshot.png
//   - build_context()      → gabung semua jadi string konteks untuk LLM prompt

use std::process::Stdio;
use tokio::process::Command;

/// Data jendela aktif dari hyprctl
#[derive(Debug, serde::Deserialize, serde::Serialize, Clone)]
pub struct ActiveWindow {
    pub class: String,
    pub title: String,
    pub pid: i64,
    pub workspace: Option<WorkspaceInfo>,
}

#[derive(Debug, serde::Deserialize, serde::Serialize, Clone)]

/// Konteks lengkap lingkungan desktop yang dikumpulkan sebelum call LLM
#[derive(Debug, Clone, serde::Serialize)]
pub struct HyprlandContext {
    pub active_window_class: String,
    pub active_window_title: String,
    pub clipboard_text: Option<String>,

}

// Fungsi publik

/// Baca jendela aktif Hyprland via `hyprctl activewindow -j`
pub async fn get_active_window() -> Option<ActiveWindow> {
    let output = Command::new("hyprctl")
        .args(["activewindow", "-j"])
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .output()
        .await
        .ok()?;

    if !output.status.success() {
        return None;
    }

    serde_json::from_slice(&output.stdout).ok()
}

/// Baca isi clipboard saat ini via `wl-paste` (wl-clipboard package) — Aman untuk UTF-8/Emoji
pub async fn get_clipboard() -> Option<String> {
    let output = Command::new("wl-paste")
        .arg("--no-newline")
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .output()
        .await
        .ok()?;

    if !output.status.success() || output.stdout.is_empty() {
        return None;
    }

    String::from_utf8(output.stdout).ok().map(|s| {
        let char_count = s.chars().count();
        if char_count > 1500 {
            let truncated: String = s.chars().take(1500).collect();
            format!("{}... [dipotong]", truncated)
        } else {
            s
        }
    })
}

/// Ambil screenshot seluruh layar pertama via `grim`, simpan ke /tmp/aoi_screenshot.png
/// Tahap 3: akan dikirim ke LLM vision model (Gemini Pro Vision)
pub async fn take_screenshot() -> Option<String> {
    let path = "/tmp/aoi_screenshot.png";

    let status = Command::new("grim")
        .arg(path)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .await
        .ok()?;

    if status.success() {
        Some(path.to_string())
    } else {
        None
    }
}

/// Kumpulkan semua konteks dan format menjadi string siap inject ke prompt LLM.
/// `include_screenshot` hanya aktif jika user secara eksplisit meminta vision.
pub async fn build_context(include_screenshot: bool) -> HyprlandContext {
    let window = get_active_window().await;
    let clipboard = get_clipboard().await;
    let screenshot = if include_screenshot {
        take_screenshot().await
    } else {
        None
    };

    HyprlandContext {
        active_window_class: window
            .as_ref()
            .map(|w| w.class.clone())
            .unwrap_or_else(|| "unknown".into()),
        active_window_title: window
            .as_ref()
            .map(|w| w.title.clone())
            .unwrap_or_else(|| "unknown".into()),
        clipboard_text: clipboard,
        screenshot_path: screenshot,
    }
}

/// Format context menjadi blok teks untuk disisipi ke system/user prompt LLM
pub fn format_context_for_prompt(ctx: &HyprlandContext) -> String {
    let mut parts = Vec::new();

    parts.push(format!(
        "[Konteks Desktop]\n- Jendela aktif: {} ({})",
        ctx.active_window_title, ctx.active_window_class
    ));

    if let Some(ref clip) = ctx.clipboard_text {
        if !clip.trim().is_empty() {
            parts.push(format!("- Clipboard:\n```\n{}\n```", clip));
        }
    }

    if ctx.screenshot_path.is_some() {
        parts.push("- Screenshot layar sudah diambil dan siap dianalisis.".into());
    }

    parts.join("\n")
}

/// Kirim conversation history ke provider yang aktif, kembalikan balasan teks asisten.
pub async fn chat(
    config: &LlmConfig,
    history: &[ChatMessage],
) -> Result<String, String> {
    match config.provider {
        LlmProvider::Gemini => gemini_chat(config, history).await,
        LlmProvider::OpenAI => openai_chat(config, history).await,
        LlmProvider::Ollama => ollama_chat(config, history).await,
    }

// ────────────────────────────────────────────────────────────────────────────
// Provider: Google Gemini (generateContent REST API)
// ────────────────────────────────────────────────────────────────────────────

async fn gemini_chat(config: &LlmConfig, history: &[ChatMessage]) -> Result<String, String> {
    let api_key = config
        .api_key
        .as_deref()
        .ok_or("GEMINI_API_KEY tidak ditemukan")?;

    let model = &config.model;
    let url = format!(
        "https://generativelanguage.googleapis.com/v1beta/models/{}:generateContent?key={}",
        model, api_key
    );

    // Konversi history ke format Gemini (contents[])
    // System prompt masuk sebagai "user" turn pertama di Gemini
    let contents: Vec<serde_json::Value> = Vec::new();

    // Sisipkan system prompt sebagai instruksi model (Gemini v1beta mendukung systemInstruction)
    let body = serde_json::json!({
        "systemInstruction": {
            "parts": [{ "text": config.system_prompt }]
        },
        "contents": history.iter().map(|m| serde_json::json!({
            "role": if m.role == "assistant" { "model" } else { "user" },
            "parts": [{ "text": m.content }]
        })).collect::<Vec<_>>(),
        "generationConfig": {
            "temperature": 0.7,
            "maxOutputTokens": 1024,
        }
    });
    // Pastikan tidak ada warning unused
    drop(contents);

    let resp = http_client()
        .post(&url)
        .json(&body)
        .send()
        .await
        .map_err(|e| format!("HTTP error Gemini: {}", e))?;

    if !resp.status().is_success() {
        let status = resp.status();
        let text = resp.text().await.unwrap_or_default();
        return Err(format!("Gemini API error {}: {}", status, text));
    }

    let raw_text = resp
        .text()
        .await
        .map_err(|e| format!("Gagal baca body Gemini: {}", e))?;

    let json: serde_json::Value = serde_json::from_str(&raw_text).map_err(|e| {
        let preview = if raw_text.len() > 150 {
            format!("{}...", &raw_text[..150])
        } else {
            raw_text.clone()
        };
        format!("Gagal parse JSON Gemini ({}): {}", e, preview)
    })?;

    let text = json["candidates"][0]["content"]["parts"][0]["text"]
        .as_str()
        .unwrap_or("")
        .to_string();

    if text.is_empty() {
        Err("Gemini mengembalikan respons kosong".into())
    } else {
        Ok(text)
    }
}

// ────────────────────────────────────────────────────────────────────────────
// Provider: OpenAI-compatible (OpenAI / Groq / OpenRouter / Custom — format identik)
// ────────────────────────────────────────────────────────────────────────────

async fn openai_chat(config: &LlmConfig, history: &[ChatMessage]) -> Result<String, String> {
    let raw_base = config
        .base_url
        .as_deref()
        .unwrap_or("https://api.openai.com/v1")
        .trim()
        .trim_end_matches('/');

    let url = if raw_base.ends_with("/chat/completions") {
        raw_base.to_string()
    } else {
        format!("{}/chat/completions", raw_base)
    };

    // Sisipkan system message di awal
    let mut messages = vec![serde_json::json!({
        "role": "system",
        "content": config.system_prompt
    })];
    for m in history {
        messages.push(serde_json::json!({
            "role": m.role,
            "content": m.content
        }));
    }

    let body = serde_json::json!({
        "model": config.model,
        "messages": messages,
        "temperature": 0.7,
        "max_tokens": 1024,
    });

    let mut req = http_client().post(&url);

    // Kirim bearer auth jika API key tersedia dan tidak kosong
    if let Some(key) = config.api_key.as_deref() {
        if !key.trim().is_empty() {
            req = req.bearer_auth(key.trim());
        }
    }

    let resp = req
        .json(&body)
        .send()
        .await
        .map_err(|e| format!("HTTP error ({}): {}", url, e))?;

    if !resp.status().is_success() {
        let status = resp.status();
        let text = resp.text().await.unwrap_or_default();
        return Err(format!("OpenAI API error {}: {}", status, text));
    }

    let raw_text = resp
        .text()
        .await
        .map_err(|e| format!("Gagal membaca respons body: {}", e))?;

    // 1. Dukungan Server-Sent Events (SSE Streaming: baris data: {"choices":[{"delta":...}]})
    // Banyak server proxy / custom AI (Next.js AI SDK, LiteLLM, Ollama proxy) selalu mengirimkan format SSE
    if raw_text.contains("data:") {
        let mut full_text = String::new();
        for line in raw_text.lines() {
            let trimmed = line.trim();
            if let Some(data_str) = trimmed.strip_prefix("data:") {
                let data_str = data_str.trim();
                if data_str == "[DONE]" || data_str.is_empty() {
                    continue;
                }
                if let Ok(chunk) = serde_json::from_str::<serde_json::Value>(data_str) {
                    if let Some(content) = chunk["choices"][0]["delta"]["content"].as_str() {
                        full_text.push_str(content);
                    } else if let Some(content) = chunk["choices"][0]["message"]["content"].as_str() {
                        full_text.push_str(content);
                    } else if let Some(text) = chunk["choices"][0]["text"].as_str() {
                        full_text.push_str(text);
                    }
                }
            }
        }
        if !full_text.is_empty() {
            return Ok(full_text);
        }
    }

    // 2. Parse sebagai JSON standar (Non-streaming)
    match serde_json::from_str::<serde_json::Value>(&raw_text) {
        Ok(json) => {
            if let Some(content) = json["choices"][0]["message"]["content"].as_str() {
                if !content.is_empty() {
                    return Ok(content.to_string());
                }
            }
            if let Some(text) = json["choices"][0]["text"].as_str() {
                if !text.is_empty() {
                    return Ok(text.to_string());
                }
            }
            if let Some(err_msg) = json["error"]["message"].as_str() {
                return Err(format!("Error dari API: {}", err_msg));
            }
            Err(format!("Format JSON response tidak dikenali: {}", raw_text))
        }
        Err(e) => {
            let preview = if raw_text.len() > 150 {
                format!("{}...", &raw_text[..150])
            } else {
                raw_text
            };
            Err(format!("Gagal parse respons JSON ({}): {}", e, preview))
        }
    }
}

/// Short-term conversation memory — dipakai penuh di Tahap 4 (persistent memory + context window)
#[allow(dead_code)]
pub struct ConversationMemory {
    messages: Vec<ChatMessage>,
    max_messages: usize,
}

#[allow(dead_code)]
impl ConversationMemory {
    pub fn new() -> Self {
        Self {
            messages: Vec::new(),
            max_messages: 20,
        }
    }

    pub fn push_user(&mut self, content: String) {
        self.messages.push(ChatMessage {
            role: "user".into(),
            content,
        });
        self.trim();
    }

    pub fn push_assistant(&mut self, content: String) {
        self.messages.push(ChatMessage {
            role: "assistant".into(),
            content,
        });
        self.trim();
    }

    pub fn messages(&self) -> &[ChatMessage] {
        &self.messages
    }

    pub fn clear(&mut self) {
        self.messages.clear();
    }

    fn trim(&mut self) {
        // Hapus pesan terlama (tapi jaga selalu berpasangan user-assistant)
        while self.messages.len() > self.max_messages {
            self.messages.remove(0);
        }
    }
}

impl Default for ConversationMemory {
    fn default() -> Self {
        Self::new()
    }
}

// Protokol:

// ────────────────────────────────────────────────────────────────────────────
// Socket Server
// ────────────────────────────────────────────────────────────────────────────

pub async fn spawn_unix_socket_listener(
    app: AppHandle,
    shared: Arc<Mutex<StatePayload>>,
    tts_player: Arc<Mutex<Option<Child>>>,
    audio_recorder: Arc<Mutex<Option<Child>>>,
) {
    if Path::new(SOCKET_PATH).exists() {
        let _ = tokio::fs::remove_file(SOCKET_PATH).await;
    }

    let listener =
        UnixListener::bind(SOCKET_PATH).expect("Gagal bind Unix Socket /tmp/ai-assistant.sock");

    println!("🚀 IPC Socket aktif di: {}", SOCKET_PATH);

    loop {
        match listener.accept().await {
            Ok((stream, _)) => {
                let app = app.clone();
                let shared = shared.clone();
                let tts_player = tts_player.clone();
                let audio_recorder = audio_recorder.clone();

                tokio::spawn(async move {
                    let reader = BufReader::new(stream);
                    let mut lines = reader.lines();
                    while let Ok(Some(line)) = lines.next_line().await {
                        handle_command(
                            line.trim(),
                            &app,
                            &shared,
                            &tts_player,
                            &audio_recorder,
                        )
                        .await;
                    }
                });
            }
            Err(e) => eprintln!("Socket accept error: {}", e),
        }
    }
}

// ────────────────────────────────────────────────────────────────────────────
// Command Handler
// ────────────────────────────────────────────────────────────────────────────

async fn handle_command(
    cmd: &str,
    app: &AppHandle,
    shared: &Arc<Mutex<StatePayload>>,
    tts_player: &Arc<Mutex<Option<Child>>>,
    audio_recorder: &Arc<Mutex<Option<Child>>>,
) {
    match cmd {
        "toggle_mic" => {
            handle_smart_toggle(app, shared, tts_player, audio_recorder).await;
        }

        "toggle_hide" => {
            shared.lock().await.minimized ^= true;
            broadcast_state(app, shared).await;
        }

        "confirm_yes" => {
            let pending = {
                let mut s = shared.lock().await;
                if s.state == AssistantState::Confirm {
                    let cmd = s.pending_command.take();
                    s.state = AssistantState::Thinking;
                    s.message = "✅ Menjalankan perintah...".into();
                    cmd
                } else {
                    None
                }
            };
            broadcast_state(app, shared).await;

            if let Some(bash_cmd) = pending {
                let result = bash_executor::execute_green(&bash_cmd).await;
                let mut s = shared.lock().await;
                match result {
                    Ok(out) => {
                        s.state = AssistantState::Speaking;
                        s.message = format!("✅ Selesai:\n{}", out.trim());
                        let app_h = app.clone();
                        let state_arc = shared.clone();
                        let tts_arc = tts_player.clone();
                        tokio::spawn(async move {
                            tts::speak(
                                "Perintah berhasil dieksekusi.".into(),
                                app_h,
                                state_arc,
                                tts_arc,
                            )
                            .await;
                        });
                    }
                    Err(e) => {
                        s.state = AssistantState::Error;
                        s.message = format!("❌ Error: {}", e);
                    }
                }
                drop(s);
                broadcast_state(app, shared).await;
            }
        }

        "confirm_no" => {
            let mut s = shared.lock().await;
            if s.state == AssistantState::Confirm {
                s.state = AssistantState::Idle;
                s.pending_command = None;
                s.message = "❌ Perintah dibatalkan oleh pengguna.".into();
            }
            drop(s);
            broadcast_state(app, shared).await;
        }

        // Testing manual state via CLI
        _ if cmd.starts_with("state:") => {
            let target = cmd.trim_start_matches("state:");
            let mut s = shared.lock().await;
            match target {
                "idle" => {
                    s.state = AssistantState::Idle;
                    s.message = "Standby... Tekan Super+V untuk bicara.".into();
                }
                "listening" => {
                    s.state = AssistantState::Listening;
                    s.message = "Mendengarkan suaramu...".into();
                }
                "thinking" => {
                    s.state = AssistantState::Thinking;
                    s.message = "Sedang memikirkan jawaban...".into();
                }
                "speaking" => {
                    s.state = AssistantState::Speaking;
                    s.message = "Halo! Ada yang bisa aku bantu di CachyOS?".into();
                }
                "confirm" => {
                    s.state = AssistantState::Confirm;
                    s.message = "Izinkan saya menjalankan perintah ini?".into();
                    s.pending_command = Some("sudo pacman -Syu".into());
                }
                "error" => {
                    s.state = AssistantState::Error;
                    s.message = "Waduh, koneksi API gagal!".into();
                }
                _ => {}
            }
            drop(s);
            broadcast_state(app, shared).await;
        }

        _ if cmd.starts_with("chat:") => {
            let user_text = cmd.trim_start_matches("chat:").trim().to_string();
            if !user_text.is_empty() {
                let app_clone = app.clone();
                let shared_clone = shared.clone();
                let tts_clone = tts_player.clone();
                tokio::spawn(async move {
                    println!("💬 Chat via socket diterima: {}", user_text);
                    let res = commands::run_ai_pipeline(
                        user_text,
                        app_clone,
                        shared_clone,
                        tts_clone,
                    )
                    .await;

pub async fn handle_smart_toggle(

    tts_player: &Arc<Mutex<Option<Child>>>,
    audio_recorder: &Arc<Mutex<Option<Child>>>,
) {

    match current_state {

                        s.state = AssistantState::Thinking;

                    }

                    let app_clone = app.clone();
                    let shared_clone = shared.clone();
                    let tts_clone = tts_player.clone();

                    tokio::spawn(async move {
                        let cfg = config::load_config();
                        match stt::transcribe(&wav_path, &cfg).await {
                            Ok(user_text) => {
                                println!("Transkripsi: {}", user_text);
                                let _ = commands::run_ai_pipeline(
                                    user_text,
                                    app_clone,
                                    shared_clone,
                                    tts_clone,
                                )
                                .await;
                            }
                            Err(e) => {
                                eprintln!("STT error: {}", e);
                                let mut s = shared_clone.lock().await;
                                s.state = AssistantState::Error;
                                s.message = format!("❌ STT: {}", e);
                                drop(s);
                                broadcast_state(&app_clone, &shared_clone).await;
                            }
                        }
                    });
                }
                Err(e) => {
                    let mut s = shared.lock().await;
                    s.state = AssistantState::Idle;
                    s.message = format!("Mikrofon standby: {}", e);
                    drop(s);
                    broadcast_state(app, shared).await;
                }
            }
        }

        // ── BARGE-IN (Interupsi): Langsung hentikan TTS dan mulai rekam ulang
        AssistantState::Thinking | AssistantState::Speaking => {
            tts::stop(tts_player).await;

            match stt::start_recording(audio_recorder).await {
                Ok(()) => {
                    let mut s = shared.lock().await;
                    s.state = AssistantState::Listening;
                    s.message = "⚡ Interupsi! Silakan bicara...".into();
                }
                Err(e) => {
                    let mut s = shared.lock().await;
                    s.state = AssistantState::Error;
                    s.message = format!("❌ Gagal mulai rekam: {}", e);
                }
            }
            broadcast_state(app, shared).await;
        }

        // ── Batalkan konfirmasi jika menekan Super+V saat confirm
        AssistantState::Confirm => {
            let mut s = shared.lock().await;
            s.state = AssistantState::Idle;
            s.pending_command = None;
            s.message = "Eksekusi perintah dibatalkan.".into();
            drop(s);
            broadcast_state(app, shared).await;
        }
    }
}

/// Mulai merekam mikrofon ke 16kHz mono 16-bit WAV (format standar Whisper)
pub async fn start_recording(recorder: &Arc<Mutex<Option<Child>>>) -> Result<(), String> {
    let mut guard = recorder.lock().await;

    // Jika sedang merekam, hentikan dulu yang lama
    if let Some(mut existing) = guard.take() {
        let _ = existing.kill().await;
    }

    // Bersihkan file lama jika ada
    if Path::new(MIC_WAV_PATH).exists() {
        let _ = tokio::fs::remove_file(MIC_WAV_PATH).await;
    }

    // 1. Coba pw-record (PipeWire natif CachyOS)
    if let Ok(child) = Command::new("pw-record")
        .args([
            "--rate", "16000",
            "--channels", "1",
            "--format", "s16",
            MIC_WAV_PATH,
        ])
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
    {
        *guard = Some(child);
        return Ok(());
    }

    // 2. Fallback: parec
    if let Ok(child) = Command::new("parec")
        .args([
            "--rate=16000",
            "--channels=1",
            "--format=s16le",
            "--file-format=wav",
            MIC_WAV_PATH,
        ])
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
    {
        *guard = Some(child);
        return Ok(());
    }

    // 3. Fallback: arecord (ALSA)
    if let Ok(child) = Command::new("arecord")
        .args([
            "-r", "16000",
            "-c", "1",
            "-f", "S16_LE",
            "-t", "wav",
            MIC_WAV_PATH,
        ])
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
    {
        *guard = Some(child);
        return Ok(());
    }

    Err("Tidak ditemukan alat perekam audio (pw-record / parec / arecord).".into())
}

/// Hentikan perekaman dengan SIGINT agar header WAV ditulis dengan bersih oleh PipeWire
pub async fn stop_recording(recorder: &Arc<Mutex<Option<Child>>>) -> Result<String, String> {
    let mut guard = recorder.lock().await;
    let mut child = guard.take().ok_or("Tidak ada sesi perekaman yang aktif.")?;

    // Kirim SIGINT (interrupt) menggunakan command `kill -s INT <pid>` agar pw-record menutup header WAV
    if let Some(pid) = child.id() {
        let _ = Command::new("kill")
            .args(["-s", "INT", &pid.to_string()])
            .status()
            .await;
    }

    // Tunggu maksimal 600ms agar proses flush & keluar secara alami, jika belum berhenti baru kill()
    match tokio::time::timeout(tokio::time::Duration::from_millis(600), child.wait()).await {
        Ok(_) => {}
        Err(_) => {
            let _ = child.kill().await;
            let _ = child.wait().await;
        }
    }

    // Beri jeda 50ms agar filesystem selesai menulis
    tokio::time::sleep(tokio::time::Duration::from_millis(50)).await;

    if !Path::new(MIC_WAV_PATH).exists() {
        return Err("File audio /tmp/aoi_mic.wav tidak berhasil dibuat.".into());
    }

    let meta = tokio::fs::metadata(MIC_WAV_PATH)
        .await
        .map_err(|e| e.to_string())?;

    if meta.len() < 100 {
        return Err("Audio rekaman kosong atau terlalu pendek.".into());
    }

    Ok(MIC_WAV_PATH.to_string())
}

/// Deteksi binary whisper di sistem
pub async fn find_whisper_binary() -> Option<&'static str> {
    let candidates = ["whisper-cli", "whisper-cpp", "whisper-main", "whisper"];
    for bin in candidates {

pub fn expand_user_path(raw: &str) -> PathBuf {
    let mut cleaned = raw.trim().trim_end_matches('.');
    cleaned = cleaned.trim();
    if cleaned.starts_with("~/") || cleaned == "~" {
        if let Ok(home) = std::env::var("HOME") {
            let sub = cleaned.strip_prefix("~/").unwrap_or("");
            return PathBuf::from(home).join(sub);
        }
    }
    PathBuf::from(cleaned)
}

/// Cari model ggml whisper
pub fn find_whisper_model(custom_path: &str) -> Option<PathBuf> {
    // 1. Path kustom dari config
    if !custom_path.is_empty() {
        let p = expand_user_path(custom_path);
        if p.exists() {
            return Some(p);
        }
    }

    // 2. Env variable
    if let Ok(env_p) = std::env::var("AOI_WHISPER_MODEL") {
        let p = expand_user_path(&env_p);
        if p.exists() {
            return Some(p);
        }
    }

    // 3. ~/.local/share/whisper/*.bin
    if let Ok(home) = std::env::var("HOME") {
        let dir = PathBuf::from(&home).join(".local/share/whisper");
        if let Some(m) = first_bin_in(&dir) {
            return Some(m);
        }
    }

    // 4. /usr/share/whisper-cpp/*.bin atau /usr/share/whisper/*.bin
    for sys in &["/usr/share/whisper-cpp", "/usr/share/whisper"] {
        if let Some(m) = first_bin_in(&PathBuf::from(sys)) {
            return Some(m);
        }
    }

    None
}

fn first_bin_in(dir: &PathBuf) -> Option<PathBuf> {
    let entries = std::fs::read_dir(dir).ok()?;
    let mut models: Vec<PathBuf> = entries
        .flatten()
        .map(|e| e.path())
        .filter(|p| {
            let ext = p.extension().and_then(|e| e.to_str());
            ext == Some("bin") || ext == Some("ggml")
        })
        .collect();
    models.sort();
    models.into_iter().next()
}

    Command::new("which")
        .arg(bin)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .await
        .map(|s| s.success())
        .unwrap_or(false)
}

/// Transkripsi file audio menjadi teks.
/// Mencoba whisper.cpp lokal terlebih dahulu; jika tidak tersedia,
/// mencoba Cloud Whisper API (Groq / OpenAI) jika API key tersedia.
pub async fn transcribe(wav_path: &str, config: &AppConfig) -> Result<String, String> {
    // ── 1. Coba Whisper.cpp Lokal
    if let Some(bin) = find_whisper_binary().await {
        if let Some(model) = find_whisper_model(&config.whisper_model_path) {
            let output = Command::new(bin)
                .args([
                    "-m", model.to_str().unwrap_or(""),
                    "-f", wav_path,
                    "--no-timestamps",
                ])
                .stdout(Stdio::piped())
                .stderr(Stdio::null())
                .output()
                .await
                .map_err(|e| format!("Gagal menjalankan whisper: {}", e))?;

            if output.status.success() {
                let text = String::from_utf8_lossy(&output.stdout)
                    .lines()
                    .map(|l| l.trim())
                    .filter(|l| !l.is_empty())
                    .collect::<Vec<_>>()
                    .join(" ");

                if !text.is_empty() {
                    return Ok(text);
                }
            }
        }
    }

    // ── 2. Fallback: Cloud Whisper (Groq / OpenAI)
    if !config.groq_api_key.is_empty() {
        return transcribe_groq(wav_path, &config.groq_api_key).await;
    }

    if !config.openai_api_key.is_empty() {
        return transcribe_openai(wav_path, &config.openai_api_key).await;
    }

    Err(
        "Whisper STT lokal belum terpasang dan API Key Cloud belum dikonfigurasi.\n\
         Silakan pasang whisper-cpp (sudo pacman -S whisper-cpp) dan unduh model ggml ke ~/.local/share/whisper/,\n\
         atau isi GROQ_API_KEY di pengaturan untuk STT awan ultra-cepat."
            .into(),
    )
}

/// Transkripsi via Groq Whisper API (whisper-large-v3, latensi ~150ms)
async fn transcribe_groq(wav_path: &str, api_key: &str) -> Result<String, String> {
    let client = reqwest::Client::new();
    let audio_bytes = tokio::fs::read(wav_path)
        .await
        .map_err(|e| format!("Gagal membaca file audio: {}", e))?;

    let file_part = reqwest::multipart::Part::bytes(audio_bytes)
        .file_name("audio.wav")
        .mime_str("audio/wav")
        .map_err(|e| e.to_string())?;

    let form = reqwest::multipart::Form::new()
        .text("model", "whisper-large-v3")
        .part("file", file_part);

    let resp = client
        .post("https://api.groq.com/openai/v1/audio/transcriptions")
        .bearer_auth(api_key)
        .multipart(form)
        .send()
        .await
        .map_err(|e| format!("HTTP request error ke Groq: {}", e))?;

    if !resp.status().is_success() {
        let status = resp.status();
        let err_txt = resp.text().await.unwrap_or_default();
        return Err(format!("Groq Whisper error {}: {}", status, err_txt));
    }

    let json: serde_json::Value = resp.json().await.map_err(|e| e.to_string())?;
    let text = json["text"].as_str().unwrap_or("").trim().to_string();

    if text.is_empty() {
        Err("Suara tidak terdengar jelas.".into())
    } else {
        Ok(text)
    }
}

// Text Cleaner: Bersihkan markdown / code block sebelum dikirim ke TTS

pub fn clean_text_for_tts(raw: &str) -> String {
    // 0. Singkirkan tag thinking jika ada
    let without_thinking = crate::commands::strip_thinking_tags(raw);

    // 1. Singkirkan baris BASH:
    let without_bash = crate::bash_executor::strip_bash_command(&without_thinking);

    // 2. Singkirkan blok kode ```...```
    let mut cleaned = String::new();
    let mut in_code_block = false;
    for line in without_bash.lines() {
        let trimmed = line.trim();
        if trimmed.starts_with("```") {
            in_code_block = !in_code_block;
            continue;
        }
        if !in_code_block {
            cleaned.push_str(line);
            cleaned.push(' ');
        }
    }

    // 3. Singkirkan markdown markup (*, #, `, >, ~, [], ())
    let mut sanitized = String::new();
    for c in cleaned.chars() {
        if matches!(c, '*' | '#' | '`' | '~' | '>' | '[' | ']') {
            continue;
        }
        sanitized.push(c);
    }

    let result = sanitized.trim();
    if result.is_empty() {
        "Perintah berhasil dieksekusi.".to_string()
    } else {
        result.to_string()
    }
}

// ────────────────────────────────────────────────────────────────────────────
// Lip-Sync: Parse WAV PCM → RMS per chunk → jadwal (delay_ms, is_mouth_open)
// ────────────────────────────────────────────────────────────────────────────

/// Hitung jadwal lip-sync dari WAV PCM data.
/// Returns: Vec<(delay_ms_dari_awal, is_mouth_open)>
/// chunk_ms: durasi tiap frame animasi (default 140ms)
pub fn compute_lipsync_schedule(wav: &[u8], chunk_ms: u64) -> Vec<(u64, bool)> {
    if wav.len() < 44 {
        return vec![];
    }

    let sample_rate =
        u32::from_le_bytes([wav[24], wav[25], wav[26], wav[27]]) as u64;
    let bits_per_sample =
        u16::from_le_bytes([wav[34], wav[35]]) as u64;
    let bytes_per_sample = (bits_per_sample / 8).max(1);

    if sample_rate == 0 {
        return vec![];
    }

    let pcm = &wav[44..];
    let samples_per_chunk = ((sample_rate * chunk_ms) / 1000) as usize;
    let bytes_per_chunk = (samples_per_chunk * bytes_per_sample as usize).max(2);

    let threshold = match bits_per_sample {
        16 => 1100.0f64,
        8  => 12.0,
        _  => 1100.0,
    };

    let mut out = Vec::new();
    let mut delay = 0u64;

    for chunk in pcm.chunks(bytes_per_chunk) {
        let rms = match bits_per_sample {
            16 => {
                let sum: f64 = chunk
                    .chunks_exact(2)
                    .map(|b| {
                        let s = i16::from_le_bytes([b[0], b[1]]) as f64;
                        s * s
                    })
                    .sum();
                let n = (chunk.len() / 2).max(1);
                (sum / n as f64).sqrt()
            }
            _ => {
                let sum: f64 = chunk
                    .iter()
                    .map(|&b| {
                        let s = b as f64 - 128.0;
                        s * s
                    })
                    .sum();
                (sum / chunk.len().max(1) as f64).sqrt()
            }
        };

        out.push((delay, rms > threshold));
        delay += chunk_ms;
    }

    out
}

/// Main entry: text → TTS synthesis → PipeWire playback + lip-sync events → Idle.
pub async fn speak(
    text: String,
    app: AppHandle,
    assistant_state: Arc<Mutex<StatePayload>>,
    player_state: Arc<Mutex<Option<Child>>>,
) {
    let wav_path = "/tmp/aoi_tts.wav";
    let cleaned = clean_text_for_tts(&text);
    let config = crate::config::load_config();

    // ── 1. Sintesis audio sesuai engine yang aktif (Edge-TTS, ElevenLabs, OpenAI Speech, atau Piper)
    if let Err(e) = synthesize_speech(&cleaned, &config, wav_path).await {
        eprintln!("TTS synthesis gagal ({}), beralih ke simulated lip-sync.", e);
        simulate_speech(&cleaned, &app, &assistant_state, &player_state).await;
        return;
    }

    // ── 3. Baca WAV + hitung jadwal lip-sync dari amplitudo RMS
    let wav_data = match tokio::fs::read(wav_path).await {
        Ok(d) => d,
        Err(e) => {
            eprintln!("Gagal baca WAV: {}", e);
            finish_speech(&app, &assistant_state).await;
            return;
        }
    };

    let schedule = compute_lipsync_schedule(&wav_data, 140);

    // ── 4. Mulai playback audio PipeWire
    let player = match start_playback(wav_path).await {
        Ok(p) => p,
        Err(e) => {
            eprintln!("Audio playback error: {}", e);
            finish_speech(&app, &assistant_state).await;
            return;
        }
    };

    *player_state.lock().await = Some(player);

    // ── 5. Emit real-time lip-sync events mengikuti RMS audio
    let start = tokio::time::Instant::now();

    for (delay_ms, is_open) in &schedule {
        let target = start + tokio::time::Duration::from_millis(*delay_ms);
        let now = tokio::time::Instant::now();
        if now < target {
            tokio::time::sleep(target - now).await;
        }

        // Jika user melakukan barge-in (Super+V), hentikan loop
        if player_state.lock().await.is_none() {
            break;
        }

        let _ = app.emit("lip-sync-update", json!({ "is_mouth_open": is_open }));
    }

    // ── 6. Tunggu proses player selesai jika belum di-kill
    {
        let mut guard = player_state.lock().await;
        if let Some(mut child) = guard.take() {
            let _ = child.wait().await;
        }
    }

    // ── 7. Tutup mulut & kembalikan state ke Idle (pertahankan pesan yang sudah tampil)
    finish_speech(&app, &assistant_state).await;
}

// ────────────────────────────────────────────────────────────────────────────
// Simulated Speech Fallback (Jika Piper/TTS belum terinstall)
// ────────────────────────────────────────────────────────────────────────────

async fn simulate_speech(
    text: &str,
    app: &AppHandle,
    assistant_state: &Arc<Mutex<StatePayload>>,
    player_state: &Arc<Mutex<Option<Child>>>,
) {
    let word_count = text.split_whitespace().count().max(1);
    // Estimasi membaca: ~160 kata/menit = ~375ms per kata
    // Dibatasi antara 1.5 detik s/d 7 detik
    let total_ms = ((word_count as u64) * 375).clamp(1500, 7000);
    let frames = total_ms / 140;

    let mut mouth_open = false;
    for i in 0..frames {
        // Cek interupsi barge-in
        {
            let s = assistant_state.lock().await;
            if s.state != AssistantState::Speaking {
                break;
            }
        }
        if player_state.lock().await.is_some() {
            // jika ada sinyal kill
        }

        // Variasi buka-tutup mulut yang lebih natural
        if i % 2 == 0 {
            mouth_open = !mouth_open;
        }
        let _ = app.emit("lip-sync-update", json!({ "is_mouth_open": mouth_open }));
        tokio::time::sleep(tokio::time::Duration::from_millis(140)).await;
    }

    finish_speech(app, assistant_state).await;
}

/// Selesai berbicara: tutup mulut, ubah state ke Idle, jaga pesan dialog tetap tampil
async fn finish_speech(
    app: &AppHandle,
    assistant_state: &Arc<Mutex<StatePayload>>,
) {
    let _ = app.emit("lip-sync-update", json!({ "is_mouth_open": false }));
    {
        let mut s = assistant_state.lock().await;
        if s.state == AssistantState::Speaking {
            s.state = AssistantState::Idle;
        }
    }
    broadcast_state(app, assistant_state).await;
}

/// Dispatcher utama sintesis suara berdasarkan engine yang dipilih di konfigurasi
pub async fn synthesize_speech(
    text: &str,
    config: &AppConfig,
    wav_output: &str,
) -> Result<(), String> {
    match config.tts_engine.as_str() {
        "edge_tts" => generate_edge_tts(text, config, wav_output).await,
        "elevenlabs" => generate_elevenlabs(text, config, wav_output).await,
        "fish_audio" => generate_fish_audio(text, config, wav_output).await,
        "openai_speech" => generate_openai_speech(text, config, wav_output).await,
        "piper" => {
            let model = find_piper_model()
                .ok_or_else(|| "Model Piper tidak ditemukan di ~/.local/share/piper/".to_string())?;
            generate_wav(text, &model, wav_output).await
        }
        _ => {
            // Default coba Edge-TTS terlebih dahulu, jika gagal coba Piper
            if let Ok(()) = generate_edge_tts(text, config, wav_output).await {
                Ok(())
            } else if let Some(model) = find_piper_model() {
                generate_wav(text, &model, wav_output).await
            } else {
                Err("Tidak ada TTS engine yang dapat digunakan.".into())
            }
        }
    }
}

/// Sintesis vokal via ElevenLabs Voice Cloning API
async fn generate_elevenlabs(
    text: &str,
    config: &AppConfig,
    output_wav: &str,
) -> Result<(), String> {
    let voice_id = if !config.elevenlabs_voice_id.trim().is_empty() {
        config.elevenlabs_voice_id.trim().to_string()
    } else if !config.active_cloned_voice_id.is_empty() {
        if let Ok(voices) = crate::voice_clone::list_cloned_voices() {
            voices.into_iter()
                .find(|v| v.id == config.active_cloned_voice_id)
                .map(|v| v.remote_voice_id)
                .unwrap_or_default()
        } else {
            String::new()
        }
    } else {
        String::new()
    };

    if voice_id.is_empty() || config.elevenlabs_api_key.trim().is_empty() {
        return Err("ElevenLabs API Key atau Voice ID belum dikonfigurasi di Pengaturan.".into());
    }

    let url = format!("https://api.elevenlabs.io/v1/text-to-speech/{}", voice_id);

    let client = reqwest::Client::new();
    let body = json!({
        "text": text,
        "model_id": "eleven_multilingual_v2",
        "voice_settings": {
            "stability": 0.5,
            "similarity_boost": 0.8
        }
    });

    let resp = client
        .post(&url)
        .header("xi-api-key", config.elevenlabs_api_key.trim())
        .header("Content-Type", "application/json")
        .header("Accept", "audio/mpeg")
        .json(&body)
        .send()
        .await
        .map_err(|e| format!("Request ke ElevenLabs gagal: {}", e))?;

    if !resp.status().is_success() {
        let err_text = resp.text().await.unwrap_or_default();
        return Err(format!("ElevenLabs API Error: {}", err_text));
    }

    let bytes = resp
        .bytes()
        .await
        .map_err(|e| format!("Gagal membaca bytes dari ElevenLabs: {}", e))?;

    let mp3_path = "/tmp/aoi_eleven_temp.mp3";
    tokio::fs::write(mp3_path, &bytes)
        .await
        .map_err(|e| format!("Gagal menyimpan stream MP3 ElevenLabs: {}", e))?;

    convert_mp3_to_wav(mp3_path, output_wav).await
}

/// Konversi berkas audio MP3 ke WAV 22050Hz Mono menggunakan ffmpeg agar siap dianalisis RMS Lip-Sync

        .args([
            "-y",
            "-i",
            mp3_path,
            "-f",
            "wav",
            "-ar",
            "22050",
            "-ac",
            "1",
            wav_path,
        ])
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .await
        .map_err(|e| format!("Gagal menjalankan ffmpeg: {}", e))?;

    Ok(())
}

/// Jalankan piper dengan teks dari stdin, output ke file WAV
async fn generate_wav(text: &str, model: &PathBuf, output: &str) -> Result<(), String> {
    let home = std::env::var("HOME").unwrap_or_else(|_| "/tmp".into());
    let share_bin = PathBuf::from(&home).join(".local/share/piper/bin/piper");
    let local_bin = PathBuf::from(&home).join(".local/bin/piper");

    let bin = if share_bin.exists() {
        share_bin
    } else if local_bin.exists() {
        local_bin
    } else if which_exists("piper").await {
        PathBuf::from("piper")
    } else if which_exists("piper-tts").await {
        PathBuf::from("piper-tts")
    } else {
        return Err("Binary piper / piper-tts tidak ditemukan di ~/.local/share/piper/bin/piper atau PATH".into());
    };

    let mut child = Command::new(&bin)
        .args([
            "--model",
            model.to_str().unwrap_or(""),
            "--output_file",
            output,
        ])
        .stdin(Stdio::piped())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
        .map_err(|e| format!("Gagal menjalankan {}: {}", bin.display(), e))?;

    if let Some(mut stdin) = child.stdin.take() {
        stdin
            .write_all(text.as_bytes())
            .await
            .map_err(|e| format!("Gagal menulis ke stdin: {}", e))?;
        stdin
            .flush()
            .await
            .map_err(|e| format!("Gagal flush stdin: {}", e))?;
    }

    let status = child
        .wait()
        .await
        .map_err(|e| format!("Wait process error: {}", e))?;

    if !status.success() {
        return Err(format!("Process exit code: {:?}", status.code()));
    }

    Ok(())
}

async fn which_exists(bin: &str) -> bool {
    Command::new("which")
        .arg(bin)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .await
        .map(|s| s.success())
        .unwrap_or(false)
}

/// Mulai playback WAV — coba pw-play, paplay, lalu aplay
async fn start_playback(wav_path: &str) -> Result<Child, String> {
    // PipeWire (Primary — CachyOS default)
    if let Ok(c) = Command::new("pw-play")
        .arg(wav_path)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
    {
        return Ok(c);
    }

    // PulseAudio fallback
    if let Ok(c) = Command::new("paplay")
        .arg(wav_path)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
    {
        return Ok(c);
    }

    // ALSA fallback
    Command::new("aplay")
        .arg(wav_path)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
        .map_err(|_| {
            "Tidak ada audio player yang dapat memutar audio (install: pipewire)".into()
        })
}

    let mut guard = player_state.lock().await;
    if let Some(mut child) = guard.take() {
        let _ = child.kill().await;
    }
}

// Modul Studio Kloning Suara Kustom (Voice Cloning Studio)
// Mendukung pembuatan profil suara kloning dari file audio (.mp3, .wav, .m4a, .ogg)
// atau rekaman mikrofon langsung, terintegrasi dengan ElevenLabs VoiceLab, Fish Audio,
// dan server GPU mandiri / Google Colab GPT-SoVITS.

use std::fs;
use std::path::PathBuf;
use base64::prelude::*;
use reqwest::multipart::{Form, Part};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ClonedVoiceMeta {
    pub id: String,
    pub name: String,
    pub engine: String, // "elevenlabs" | "fish_audio" | "custom_server"
    pub remote_voice_id: String,
    pub sample_filename: Option<String>,
    pub created_at: u64,
    pub is_active: bool,
}

pub fn get_cloned_voices_dir() -> PathBuf {
    let home = std::env::var("HOME").unwrap_or_else(|_| "/tmp".into());
    PathBuf::from(home).join(".config/aoi/cloned_voices")
}

pub fn ensure_cloned_voices_initialized() {
    let dir = get_cloned_voices_dir();
    let _ = fs::create_dir_all(&dir);
}

/// Muat daftar semua profil suara hasil kloning
pub fn list_cloned_voices() -> Result<Vec<ClonedVoiceMeta>, String> {
    ensure_cloned_voices_initialized();
    let dir = get_cloned_voices_dir();
    let manifest_path = dir.join("voices_manifest.json");

    let cfg = crate::config::load_config();
    let active_id = cfg.active_cloned_voice_id;

    if manifest_path.exists() {
        if let Ok(content) = fs::read_to_string(&manifest_path) {
            if let Ok(mut list) = serde_json::from_str::<Vec<ClonedVoiceMeta>>(&content) {
                for item in &mut list {
                    item.is_active = !active_id.is_empty() && item.id == active_id;
                }
                return Ok(list);
            }
        }
    }

    Ok(Vec::new())
}

/// Simpan daftar profil suara kloning ke manifest
fn save_manifest(list: &[ClonedVoiceMeta]) -> Result<(), String> {
    let dir = get_cloned_voices_dir();
    let _ = fs::create_dir_all(&dir);
    let manifest_path = dir.join("voices_manifest.json");
    let json_str = serde_json::to_string_pretty(list)
        .map_err(|e| format!("Gagal serialisasi voices_manifest: {}", e))?;
    fs::write(&manifest_path, json_str)
        .map_err(|e| format!("Gagal menulis voices_manifest.json: {}", e))?;
    Ok(())
}

/// Aktifkan salah satu profil suara kloning
pub fn set_active_cloned_voice(voice_id: &str) -> Result<(), String> {
    let mut list = list_cloned_voices()?;
    let mut found = false;
    let mut selected_engine = String::new();
    let mut selected_remote_id = String::new();

    for item in &mut list {
        if item.id == voice_id {
            item.is_active = true;
            found = true;
            selected_engine = item.engine.clone();
            selected_remote_id = item.remote_voice_id.clone();
        } else {
            item.is_active = false;
        }
    }

    if !found && !voice_id.is_empty() {
        return Err(format!("Profil suara '{}' tidak ditemukan.", voice_id));
    }

    save_manifest(&list)?;

    // Perbarui konfigurasi utama aplikasi
    let mut cfg = crate::config::load_config();
    cfg.active_cloned_voice_id = voice_id.to_string();

    if found {
        if selected_engine == "elevenlabs" {
            cfg.tts_engine = "elevenlabs".to_string();
            cfg.elevenlabs_voice_id = selected_remote_id;
        } else if selected_engine == "fish_audio" {
            cfg.tts_engine = "fish_audio".to_string();
        } else if selected_engine == "custom_server" {
            cfg.tts_engine = "openai_speech".to_string();
        }
    }

    let _ = crate::config::save_config(&cfg);
    Ok(())
}

/// Hapus profil suara kloning
pub fn delete_cloned_voice(voice_id: &str) -> Result<(), String> {
    let mut list = list_cloned_voices()?;
    let initial_len = list.len();
    let mut sample_to_remove: Option<String> = None;

    list.retain(|item| {
        if item.id == voice_id {
            sample_to_remove = item.sample_filename.clone();
            false
        } else {
            true
        }
    });

    if list.len() == initial_len {
        return Err("Profil suara tidak ditemukan.".to_string());
    }

    // Hapus file audio sampel jika ada
    if let Some(filename) = sample_to_remove {
        let sample_path = get_cloned_voices_dir().join(&filename);
        if sample_path.exists() {
            let _ = fs::remove_file(sample_path);
        }
    }

    save_manifest(&list)?;

    // Reset config jika yang dihapus sedang aktif
    let mut cfg = crate::config::load_config();
    if cfg.active_cloned_voice_id == voice_id {
        cfg.active_cloned_voice_id = String::new();
        cfg.tts_engine = "edge_tts".to_string();
        let _ = crate::config::save_config(&cfg);
    }

    Ok(())
}

        name: name.trim().to_string(),
        engine: engine.to_string(),
        remote_voice_id: remote_voice_id.trim().to_string(),
        sample_filename: None,
        created_at: timestamp,
        is_active: true,
    };

    let mut list = list_cloned_voices()?;
    list.retain(|x| x.id != id);
    list.push(meta.clone());
    save_manifest(&list)?;

    let _ = set_active_cloned_voice(&id);

    Ok(meta)
}

/// Kloning suara baru dengan mengunggah berkas audio sampel (WAV / MP3)
pub async fn clone_voice_from_file(
    name: &str,
    engine: &str,
    file_base64: &str,
    filename: &str,
    api_key_override: Option<&str>,
) -> Result<ClonedVoiceMeta, String> {
    ensure_cloned_voices_initialized();

    if name.trim().is_empty() {
        return Err("Nama suara tidak boleh kosong.".into());
    }

    // Decode base64 audio
    let clean_b64 = if let Some(idx) = file_base64.find(";base64,") {
        &file_base64[idx + 8..]
    } else {
        file_base64
    };

    let audio_bytes = BASE64_STANDARD
        .decode(clean_b64.trim())
        .map_err(|e| format!("Gagal decode file audio base64: {}", e))?;

    if audio_bytes.is_empty() {
        return Err("File audio kosong.".into());
    }

    let timestamp = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_secs())
        .unwrap_or(0);

    // Simpan file audio sampel ke disk
    let ext = std::path::Path::new(filename)
        .extension()
        .and_then(|e| e.to_str())
        .unwrap_or("mp3")
        .to_lowercase();

    let sample_name = format!("sample_{}.{}", timestamp, ext);
    let sample_path = get_cloned_voices_dir().join(&sample_name);
    fs::write(&sample_path, &audio_bytes)
        .map_err(|e| format!("Gagal menyimpan audio sampel: {}", e))?;

    let cfg = crate::config::load_config();
    let client = reqwest::Client::new();

    let remote_voice_id = match engine {
        "elevenlabs" => {
            let key = api_key_override
                .filter(|k| !k.trim().is_empty())
                .unwrap_or(&cfg.elevenlabs_api_key);

            if key.trim().is_empty() {
                return Err("ElevenLabs API Key belum diisi di Pengaturan.".into());
            }

            let mime_type = match ext.as_str() {
                "wav" => "audio/wav",
                "ogg" => "audio/ogg",
                "m4a" => "audio/mp4",
                _ => "audio/mpeg",
            };

            let part = Part::bytes(audio_bytes)
                .file_name(filename.to_string())
                .mime_str(mime_type)
                .map_err(|e| format!("MIME type error: {}", e))?;

            let form = Form::new()
                .text("name", name.trim().to_string())
                .text("description", "Cloned via Aoi Assistant VoiceLab".to_string())
                .part("files", part);

            let resp = client
                .post("https://api.elevenlabs.io/v1/voices/add")
                .header("xi-api-key", key.trim())
                .multipart(form)
                .send()
                .await
                .map_err(|e| format!("Request ke ElevenLabs gagal: {}", e))?;

            if !resp.status().is_success() {
                let err_text = resp.text().await.unwrap_or_default();
                return Err(format!("ElevenLabs VoiceLab error: {}", err_text));
            }

            let json: serde_json::Value = resp
                .json()
                .await
                .map_err(|e| format!("Gagal parse respon JSON ElevenLabs: {}", e))?;

            json["voice_id"]
                .as_str()
                .ok_or_else(|| "ElevenLabs tidak mengembalikan voice_id".to_string())?
                .to_string()
        }
        "fish_audio" => {
            let key = api_key_override
                .filter(|k| !k.trim().is_empty())
                .unwrap_or(&cfg.fish_audio_api_key);

            if key.trim().is_empty() {
                return Err("Fish Audio API Key belum diisi di Pengaturan.".into());
            }

            let mime_type = match ext.as_str() {
                "wav" => "audio/wav",
                _ => "audio/mpeg",
            };

            let part = Part::bytes(audio_bytes)
                .file_name(filename.to_string())
                .mime_str(mime_type)
                .map_err(|e| format!("MIME error: {}", e))?;

            let form = Form::new()
                .text("title", name.trim().to_string())
                .part("voices", part);

            let resp = client
                .post("https://api.fish.audio/model")
                .header("Authorization", format!("Bearer {}", key.trim()))
                .multipart(form)
                .send()
                .await
                .map_err(|e| format!("Request ke Fish Audio gagal: {}", e))?;

            if !resp.status().is_success() {
                let err_text = resp.text().await.unwrap_or_default();
                return Err(format!("Fish Audio API error: {}", err_text));
            }

            let json: serde_json::Value = resp
                .json()
                .await
                .map_err(|e| format!("Gagal parse respon JSON Fish Audio: {}", e))?;

            json["_id"]
                .as_str()
                .or_else(|| json["id"].as_str())
                .ok_or_else(|| "Fish Audio tidak mengembalikan model ID".to_string())?
                .to_string()
        }
        "custom_server" => {
            // Gunakan path audio sampel lokal sebagai referensi zero-shot Colab / server GPU
            sample_path.to_string_lossy().to_string()
        }
        _ => {
            return Err(format!("Engine '{}' tidak didukung untuk kloning suara.", engine));
        }
    };

    let id = format!("clone_{}", timestamp);
    let meta = ClonedVoiceMeta {
        id: id.clone(),
        name: name.trim().to_string(),
        engine: engine.to_string(),
        remote_voice_id,
        sample_filename: Some(sample_name),
        created_at: timestamp,
        is_active: true,
    };

    let mut list = list_cloned_voices()?;
    list.push(meta.clone());
    save_manifest(&list)?;

    let _ = set_active_cloned_voice(&id);

    Ok(meta)
}

/// Ambil file sampel audio dalam format Base64 untuk preview di frontend
pub fn get_cloned_voice_sample_base64(voice_id: &str) -> Result<String, String> {
    let list = list_cloned_voices()?;
    let item = list
        .into_iter()
        .find(|v| v.id == voice_id)
        .ok_or_else(|| "Profil suara tidak ditemukan.".to_string())?;

    if let Some(sample_name) = item.sample_filename {
        let path = get_cloned_voices_dir().join(&sample_name);
        if path.exists() {
            let bytes = fs::read(&path)
                .map_err(|e| format!("Gagal membaca file sampel: {}", e))?;
            return Ok(BASE64_STANDARD.encode(&bytes));
        }
    }
    Err("File sampel audio tidak ditemukan di penyimpanan lokal.".to_string())
}
